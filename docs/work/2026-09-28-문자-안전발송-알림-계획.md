```python
def list_sms_templates(*, group=None, timing=None, active_only=True):
    conn = get_db()
    where, vals = [], []
    if active_only:
        where.append("active = 1")
    if group:
        where.append("template_group = ?"); vals.append(group)
    if timing:
        where.append("COALESCE(timing, '수시') = ?"); vals.append(timing)
    where_sql = ("WHERE " + " AND ".join(where)) if where else ""
    rows = conn.execute(
        f"SELECT *, COALESCE(timing, '수시') AS timing FROM sms_templates {where_sql} "
        "ORDER BY template_group, name", vals).fetchall()
    conn.close()
    return [dict(r) for r in rows]


def create_sms_template(*, name, body, template_group="공통", timing="수시") -> int:
    conn = get_db()
    cur = conn.execute(
        "INSERT INTO sms_templates (name, template_group, body, timing) VALUES (?, ?, ?, ?)",
        (name, template_group, body, timing))
    tid = cur.lastrowid
    conn.commit(); conn.close()
    return tid
```

```python
def _sms_reminder_done_keys(conn, keys):
    if not keys:
        return set()
    done = set()
    keys = list(keys)
    for i in range(0, len(keys), 900):
        chunk = keys[i:i + 900]
        ph = ",".join("?" * len(chunk))
        done |= {r[0] for r in conn.execute(
            f"SELECT reminder_key FROM sms_log WHERE reminder_key IN ({ph}) AND status IN ('sent','test','phone')", chunk)}
        done |= {r[0] for r in conn.execute(
            f"SELECT reminder_key FROM sms_reminder_closures WHERE reminder_key IN ({ph})", chunk)}
    return done


def sms_reminders(today=None):
    """오늘 보낼 안내 문자 — 대시보드 '안내 문자'와 상담 상세가 같이 쓴다(같은 숫자는 한 함수).
    시점에 사용 중 템플릿이 없으면 그 시점은 꺼진 것으로 본다. 입·퇴원은 admission_flow_events만 쓴다."""
    from config import SMS_CONSULT_FOLLOWUP_DAYS, SMS_AFTER_DISCHARGE_DAYS
    today = (today or datetime.now().strftime("%Y-%m-%d"))[:10]
    t0 = datetime.strptime(today, "%Y-%m-%d").date()
    on = {t["timing"] for t in list_sms_templates()}
    out = {}

    def add(key, timing, patient_id, consultation_id, name, phone, anchor):
        out.setdefault(key, {"key": key, "timing": timing, "patient_id": patient_id,
                             "consultation_id": consultation_id, "patient_name": name or "",
                             "guardian_phone": phone or "", "anchor_date": anchor,
                             "has_phone": bool((phone or "").strip())})

    conn = get_db()
    if "상담 직후" in on:
        since = (t0 - timedelta(days=SMS_CONSULT_FOLLOWUP_DAYS - 1)).isoformat()
        for r in conn.execute(
                """SELECT c.id, c.patient_id, c.consult_date, p.name, p.guardian_phone FROM consultations c
                   JOIN patients p ON p.id = c.patient_id
                   WHERE c.consult_date BETWEEN ? AND ? AND COALESCE(c.consult_result,'') != '상담취소'""",
                (since, today)):
            add(f"상담 직후:c{r['id']}", "상담 직후", r["patient_id"], r["id"], r["name"], r["guardian_phone"], r["consult_date"])
    if "입원 전날" in on:
        tomorrow = (t0 + timedelta(days=1)).isoformat()
        for r in conn.execute(
                """SELECT c.id, c.patient_id, p.name, p.guardian_phone FROM consultations c

                (tomorrow,)):

    phones = {}

        if pid not in phones:
            row = conn.execute("SELECT guardian_phone FROM patients WHERE id=?", (pid,)).fetchone()
            phones[pid] = row["guardian_phone"] if row else ""
        return phones[pid]
    flows = []
    if on & {"입원 당일", "퇴원 당일"}:

    after = (t0 - timedelta(days=SMS_AFTER_DISCHARGE_DAYS)).isoformat()

        if e["kind"] == "in" and day == today and "입원 당일" in on:

def close_sms_reminder(*, key, reason, closed_by, patient_id=None, consultation_id=None) -> bool:
    conn = get_db()
    cur = conn.execute(
        "INSERT OR IGNORE INTO sms_reminder_closures (reminder_key, patient_id, consultation_id, reason, closed_by) "
        "VALUES (?,?,?,?,?)", (key, patient_id, consultation_id, reason, closed_by))
    conn.commit(); conn.close()
    return cur.rowcount == 1


def reminder_status(consultation_id: int) -> dict:
    """상담 상세 한 줄 — 이 상담의 알림(보낼 것)·보낸 문자·안 보냄 사유."""
    pending = [r for r in sms_reminders() if r["consultation_id"] == consultation_id]
    conn = get_db()
    sent = [dict(r) for r in conn.execute(
        "SELECT datetime(created_at,'localtime') AS created_at, status FROM sms_log "
        "WHERE consultation_id=? AND status IN ('sent','test','phone','manual') ORDER BY created_at DESC",
        (consultation_id,))]
    closed = [dict(r) for r in conn.execute(
        "SELECT reason, datetime(created_at,'localtime') AS created_at FROM sms_reminder_closures "
        "WHERE consultation_id=? ORDER BY created_at DESC", (consultation_id,))]
    conn.close()
    return {"pending": pending, "sent": sent, "closed": closed}
```
